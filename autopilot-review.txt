import { NextRequest, NextResponse } from "next/server";
import prisma from "@/lib/db";

export const runtime = "nodejs";

export async function GET(req: NextRequest) {
  try {
    const { searchParams } = new URL(req.url);
    const where: any = {};

    const channelId = searchParams.get("channelId");
    const userId = searchParams.get("userId");
    const status = searchParams.get("status");

    if (channelId) where.channelId = channelId;
    if (userId) where.userId = userId;
    if (status) where.status = status;

    const videos = await prisma.video.findMany({
      where,
      orderBy: { createdAt: "desc" },
    });

    return NextResponse.json(videos);
  } catch (error: any) {
    return NextResponse.json(
      { error: error.message || "Failed to fetch videos" },
      { status: 500 }
    );
  }
}

export async function POST(req: NextRequest) {
  try {
    const data = await req.json();

    if (Array.isArray(data.tags)) {
      data.tags = JSON.stringify(data.tags);
    }

    const video = await prisma.video.create({ data });
    return NextResponse.json(video);
  } catch (error: any) {
    return NextResponse.json(
      { error: error.message || "Failed to create video" },
      { status: 500 }
    );
  }
}

export async function PATCH(req: NextRequest) {
  try {
    const { id, ...updates } = await req.json();

    if (Array.isArray(updates.tags)) {
      updates.tags = JSON.stringify(updates.tags);
    }

    const video = await prisma.video.update({
      where: { id },
      data: updates,
    });

    return NextResponse.json(video);
  } catch (error: any) {
    return NextResponse.json(
      { error: error.message || "Failed to update video" },
      { status: 500 }
    );
  }
}

function createFallbackThumbnail(title: string, topic = ""): string {
  try {
    const canvas = document.createElement("canvas")
    canvas.width = 1280
    canvas.height = 720

    const ctx = canvas.getContext("2d")
    if (!ctx) return ""

    const gradient = ctx.createLinearGradient(0, 0, 1280, 720)
    gradient.addColorStop(0, "#09090b")
    gradient.addColorStop(0.55, "#17111f")
    gradient.addColorStop(1, "#321014")

    ctx.fillStyle = gradient
    ctx.fillRect(0, 0, 1280, 720)

    // Decorative background elements.
    ctx.fillStyle = "rgba(239, 68, 68, 0.10)"
    ctx.beginPath()
    ctx.arc(1110, 110, 300, 0, Math.PI * 2)
    ctx.fill()

    ctx.fillStyle = "#ef4444"
    ctx.fillRect(0, 0, 14, 720)
    ctx.fillRect(64, 76, 128, 8)

    // Brand label.
    ctx.fillStyle = "#ffffff"
    ctx.font = "bold 26px Arial"
    ctx.textAlign = "left"
    ctx.fillText("YouTubeAuto.ai", 64, 130)

    ctx.fillStyle = "#f87171"
    ctx.font = "bold 20px Arial"
    ctx.fillText("VIDEO PREVIEW", 64, 190)

    // Wrap long titles without overflowing the canvas.
    const text = (title || topic || "Your next video").trim().toUpperCase()
    const words = text.split(/\s+/)
    const maxWidth = 1080
    const lineHeight = 76
    const lines: string[] = []
    let line = ""

    ctx.font = "bold 56px Arial"

    for (const word of words) {
      const candidate = line ? `${line} ${word}` : word

      if (ctx.measureText(candidate).width > maxWidth && line) {
        lines.push(line)
        line = word
      } else {
        line = candidate
      }
    }

    if (line) lines.push(line)

    const visibleLines = lines.slice(0, 4)
    let y = 300

    for (const item of visibleLines) {
      let output = item

      while (ctx.measureText(output).width > maxWidth && output.length > 3) {
        output = `${output.slice(0, -4)}...`
      }

      ctx.fillStyle = "#ffffff"
      ctx.fillText(output, 64, y)
      y += lineHeight
    }

    // Bottom accent and preview label.
    ctx.fillStyle = "rgba(255,255,255,0.12)"
    ctx.fillRect(64, 620, 1152, 1)

    ctx.fillStyle = "#ef4444"
    ctx.beginPath()
    ctx.roundRect(64, 646, 170, 42, 10)
    ctx.fill()

    ctx.fillStyle = "#ffffff"
    ctx.font = "bold 18px Arial"
    ctx.fillText("VIDEO", 111, 673)

    ctx.textAlign = "right"
    ctx.fillStyle = "#d4d4d8"
    ctx.font = "18px Arial"
    ctx.fillText("Thumbnail preview", 1216, 674)

    return canvas.toDataURL("image/jpeg", 0.88)
  } catch {
    return ""
  }
}

function VideoThumbnail({ video }: { video: VideoType }) {
  const [src, setSrc] = useState("")
  const [usingFallback, setUsingFallback] = useState(false)
  const [imageFailed, setImageFailed] = useState(false)

  useEffect(() => {
    let cancelled = false

    setSrc("")
    setUsingFallback(false)
    setImageFailed(false)

    const original =
      video.thumbnailUrl?.trim() || video.thumbnail?.trim() || ""

    // Keep the actual thumbnail whenever it exists.
    if (
      original &&
      (original.startsWith("/") ||
        original.startsWith("https://") ||
        original.startsWith("http://") ||
        original.startsWith("data:image/"))
    ) {
      setSrc(original)
      return
    }

    const fallback = createFallbackThumbnail(video.title, video.topic)

    if (!cancelled) {
      setUsingFallback(true)
      if (fallback) setSrc(fallback)
      else setImageFailed(true)
    }

    return () => {
      cancelled = true
    }
  }, [video.id, video.title, video.topic, video.thumbnail, video.thumbnailUrl])

  const handleImageError = () => {
    // A missing real thumbnail should never leave a broken-image icon.
    if (!usingFallback) {
      const fallback = createFallbackThumbnail(video.title, video.topic)
      setUsingFallback(true)

      if (fallback) {
        setSrc(fallback)
      } else {
        setImageFailed(true)
      }
      return
    }

    setImageFailed(true)
  }

  if (!src || imageFailed) {
    return (
      <div className="flex aspect-video w-full flex-col items-center justify-center gap-2 bg-gradient-to-br from-zinc-950 via-zinc-900 to-red-950 text-zinc-400">
        <ImageOff className="h-8 w-8" />
        <span className="text-xs">Thumbnail unavailable</span>
      </div>
    )
  }

  return (
    <div className="relative aspect-video w-full overflow-hidden bg-zinc-950">
      <img
        src={src}
        alt={`${video.title || "Video"} thumbnail`}
        loading="lazy"
        decoding="async"
        onError={handleImageError}
        className="h-full w-full object-cover transition-transform duration-500 group-hover:scale-[1.03]"
      />

      {usingFallback && (
        <div className="absolute bottom-2 left-2 rounded-md border border-white/10 bg-black/70 px-2 py-1 text-[10px] text-white/80 backdrop-blur">
          Preview thumbnail
        </div>
      )}

      <div className="pointer-events-none absolute inset-0 bg-gradient-to-t from-black/35 via-transparent to-transparent" />
    </div>
  )
}

function formatDate(value?: string) {
  if (!value) return "Date unavailable"

  const date = new Date(value)
  if (Number.isNaN(date.getTime())) return "Date unavailable"

  return new Intl.DateTimeFormat("en-IN", {
    day: "numeric",
    month: "short",
    year: "numeric",
  }).format(date)
}

  const [selectedChannel, setSelectedChannel] = useState("main-channel")
  const [searchQuery, setSearchQuery] = useState("")
  const [isRefreshing, setIsRefreshing] = useState(false)
  const [loaded, setLoaded] = useState(false)

  const refreshVideos = useCallback(() => {
    if (!user?.email || !store) {
      setChannels([])
      setVideos([])
      setLoaded(true)
      return
    }

    const localUser = store.getUserByEmail(user.email)

    if (!localUser) {
      setChannels([])
      setVideos([])
      setLoaded(true)
      console.warn(
        "My Videos: no local Store profile matches the signed-in email."
      )
      return
    }

    store.repairOrphanedChannelsForSingleUser(localUser.id)

    const userChannels = store.getChannels(localUser.id)
    setChannels(userChannels)

    const channelIds = new Set(userChannels.map((channel) => channel.id))

    const userVideos = store
      .getVideos()
      .filter((video) => channelIds.has(video.channelId))
      .sort((a, b) => {
        const dateA = new Date(a.createdAt || 0).getTime()
        const dateB = new Date(b.createdAt || 0).getTime()
        return dateB - dateA
      })

    setVideos(userVideos)
    setLoaded(true)
  }, [user?.email])

  useEffect(() => {
    refreshVideos()

    const handleStorage = () => refreshVideos()
    const handleFocus = () => refreshVideos()

    window.addEventListener("storage", handleStorage)
    window.addEventListener("focus", handleFocus)

    return () => {
      window.removeEventListener("storage", handleStorage)
      window.removeEventListener("focus", handleFocus)
    }
  }, [refreshVideos])

  const counts = useMemo(
    () => ({
      all: videos.length,
      live: videos.filter((video) => video.status === "live").length,
      pending: videos.filter((video) =>
        ["pending-approval", "approved", "uploading", "user-approved"].includes(
          video.status
        )
      ).length,
      failed: videos.filter((video) =>
        ["failed", "rejected"].includes(video.status)
      ).length,
    }),
    [videos]
  )

      !fileName ||

      fileName === ".." ||
      !/\.(mp4|mov|webm|m4v)$/i.test(fileName)
    ) {
      return jsonError("Invalid video filename or format.", 400);
    }

      return jsonError(
        "Video file missing on server. Generate or save the video first.",
        404

      return jsonError("Video file is empty or invalid.", 400);

        "uploadLimitExceeded",
        "quotaExceeded",
        "dailyLimitExceeded",

          process.cwd(),
          "storage",
          "ready_to_upload"

              title: title.trim(),
              description,
              tags,
              categoryId,
              privacyStatus: privacy,
              scheduledTime: scheduledTime || null,

          queued: true,
          status: "queued",

        status: initResponse.status,

            controller.close();
          } else {
            controller.enqueue(value);
          }
        } catch (error) {
          controller.error(error);
        }

        fileStream.destroy();

        headers: {
          "Content-Type": "video/mp4",
          "Content-Length": String(videoStats.size),

        duplex: "half",
        signal: AbortSignal.timeout(300000),
      });
    } catch (error) {
      fileStream.destroy();
      throw error;
    }

    if (!uploadResponse.ok) {
      const errorText = await uploadResponse.text();

      console.error("[YouTube video upload failed]", {
        status: uploadResponse.status,
        error: errorText,
      });

      return jsonError(
        `YouTube upload failed (${uploadResponse.status}): ${errorText.slice(0, 1000)}`,
        502
      );
    }

    const uploadedVideo = await uploadResponse.json();

    if (!uploadedVideo?.id) {
      return jsonError(
        "YouTube did not return a video ID. Upload status needs verification.",
        502
      );
    }

    // 7. Clean up old subtitle files only.
    try {
      const subtitlesDirectory = path.join(
        process.cwd(),
        "storage",
        "subtitles"
      );

      if (existsSync(subtitlesDirectory)) {

            const stats = statSync(fullPath);

            if (Date.now() - stats.mtimeMs > 60 * 60 * 1000) {
              await unlink(fullPath);
            }
          } catch {
            // Ignore individual subtitle cleanup failures.
          }
        }
      }
    } catch {
      // Subtitle cleanup must not invalidate a successful upload.
    }

    const youtubeUrl = `https://www.youtube.com/watch?v=${uploadedVideo.id}`;
    const returnedPrivacy =
      uploadedVideo.status?.privacyStatus || privacy;

    const response = NextResponse.json({
      success: true,
      queued: false,
      status: returnedPrivacy,
      videoId: uploadedVideo.id,
      youtubeUrl,
      message:
        returnedPrivacy === "public"
          ? "Upload completed. YouTube reports the video as public."
          : `Upload completed, but the video is ${returnedPrivacy}. It is not publicly live.`,
    });

    return response;
  } catch (error) {
    console.error("[YouTube upload route error]", error);

    return jsonError(
      error instanceof Error ? error.message : "Unexpected upload error.",
      500
    );
  }

async function uploadVideoMemorySafe(videoPath, metadata, token) {
  const fileSize = statSync(videoPath).size;
  
  const initRes = await fetch(
    "https://www.googleapis.com/upload/youtube/v3/videos?uploadType=resumable&part=snippet,status",
    {
      method: "POST",
      headers: {
        Authorization: `Bearer ${token}`,
        "Content-Type": "application/json",
        "X-Upload-Content-Type": "video/mp4",
        "X-Upload-Content-Length": fileSize.toString(),
      },
      body: JSON.stringify({
        snippet: {
          title: (metadata?.title || "Video").slice(0, 100),
          description: (metadata?.description || "Generated by YouTubeAuto.ai") + "\n\n#viral #trending",
          tags: Array.isArray(metadata?.tags) ? metadata.tags.slice(0, 15) : [],
          categoryId: metadata?.categoryId || "22"
        },
        status: { privacyStatus: metadata?.privacyStatus || "public", selfDeclaredMadeForKids: false },
      }),
    }
  );

  if (!initRes.ok) {
    const err = await initRes.json().catch(() => ({}));
    if (err?.error?.errors?.[0]?.reason === "uploadLimitExceeded") throw new Error("QUOTA_EXCEEDED");
    throw new Error(err?.error?.message || "Init failed: " + initRes.status);
  }

  const uploadUrl = initRes.headers.get("location");
  if (!uploadUrl) throw new Error("No upload URL returned from YouTube");

  // STREAM-BASED UPLOAD (Memory Safe - No RAM Overload)
  const fileStream = createReadStream(videoPath, { highWaterMark: 1 * 1024 * 1024 });
  const iterator = fileStream[Symbol.asyncIterator]();
  const stream = new ReadableStream({
    async pull(controller) {
      const { value, done } = await iterator.next();
      if (done) { controller.close(); } else { controller.enqueue(value); }
    }
  });

  const uploadRes = await fetch(uploadUrl, {
    method: "PUT",
    headers: { "Content-Type": "video/mp4", "Content-Length": fileSize.toString() },
    body: stream,
    duplex: 'half'
  });

  if (!uploadRes.ok) {
    const errText = await uploadRes.text();
    throw new Error("Stream upload failed: " + errText);
  }

  const uploadedData = await uploadRes.json();
  return uploadedData.id;
}

export async function GET() {
  try {
    if (!existsSync(QUEUE_DIR)) return NextResponse.json({ queued: 0, items: [] });
    const files = (await readdir(QUEUE_DIR)).filter(f => f.endsWith(".json"));
    const items = [];
    for (const f of files) {
      try {
        const d = JSON.parse(await readFile(path.join(QUEUE_DIR, f), "utf8"));
        items.push({ file: f, title: d.title, queuedAt: d.queuedAt, videoUrl: d.videoUrl });
      } catch {}
    }
    return NextResponse.json({ queued: items.length, items });
  } catch(e: any) {
    return NextResponse.json({ error: e.message }, { status: 500 });
  }
}

export async function POST(req) {
  try {
    const { processAll = false, accessToken } = await req.json().catch(() => ({}));
    if (!existsSync(QUEUE_DIR)) return NextResponse.json({ processed: 0, message: "Queue empty" });

    const files = (await readdir(QUEUE_DIR)).filter(f => f.endsWith(".json"));
    if (files.length === 0) return NextResponse.json({ processed: 0, message: "Queue empty" });

    const token = accessToken || process.env.YOUTUBE_ACCESS_TOKEN;
    if (!token) return NextResponse.json({ error: "No YouTube token available. Connect YouTube first." }, { status: 401 });

    const toProcess = processAll ? files : files.slice(0, 1);
    const results = [];

    for (const file of toProcess) {
      const jsonPath = path.join(QUEUE_DIR, file);
      try {
        const queuedData = JSON.parse(await readFile(jsonPath, "utf8"));
        const fileName = queuedData.videoUrl.split("/").pop();
        const videoPath = path.join(process.cwd(), "storage", "videos", fileName);

        if (!existsSync(videoPath)) {
          await unlink(jsonPath);
          results.push({ file, status: "skipped", reason: "video file missing" });
          continue;
        }

        console.log(`?? Retrying upload for: ${queuedData.title}`);
        const videoId = await uploadVideoMemorySafe(videoPath, queuedData, token);
        await unlink(jsonPath);
        results.push({ file, status: "uploaded", videoId, url: `https://youtube.com/watch?v=${videoId}` });
        console.log(`? Retry Successful! Video ID: ${videoId}`);
        
        if (toProcess.length > 1) await new Promise(r => setTimeout(r, 3000));
      } catch(e: any) {
        if (e.message === "QUOTA_EXCEEDED") {
          results.push({ file, status: "quota_exceeded" });
          break;
        }
        results.push({ file, status: "failed", reason: e.message });
      }
    }

    return NextResponse.json({ success: true, processed: results.length, results });

  } catch(error: any) {
    console.error("Retry Queue Error:", error.message);
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}

type CronParts = { minute: number; hour: number; day: number; month: number; weekday: number };

function getTimeParts(date: Date, timeZone: string): CronParts {
  const parts = new Intl.DateTimeFormat("en-US", {
    timeZone, minute: "2-digit", hour: "2-digit", day: "2-digit", month: "2-digit", weekday: "short", hourCycle: "h23",
  }).formatToParts(date);
  const v = Object.fromEntries(parts.map((p) => [p.type, p.value]));
  const days: Record<string, number> = { Sun: 0, Mon: 1, Tue: 2, Wed: 3, Thu: 4, Fri: 5, Sat: 6 };
  return { minute: Number(v.minute), hour: Number(v.hour), day: Number(v.day), month: Number(v.month), weekday: days[v.weekday] ?? date.getDay() };
}

function fieldMatches(field: string, value: number, min: number, max: number): boolean {
  if (field === "*") return true;
  return field.split(",").some((part) => {
    const [base, stepText] = part.split("/");
    const step = stepText ? Number(stepText) : 1;
    if (!Number.isInteger(step) || step < 1) return false;
    let low = min, high = max;
    if (base && base !== "*") {
      if (base.includes("-")) {
        const pair = base.split("-").map(Number);
        if (pair.length !== 2) return false;
        [low, high] = pair;
      } else {
        low = Number(base); high = low;
      }
    }
    return Number.isInteger(low) && Number.isInteger(high) && low >= min && high <= max && value >= low && value <= high && (value - low) % step === 0;
  });
}

function matchesSchedule(expression: string, date: Date, timeZone: string): boolean {
  const fields = expression.trim().split(/\s+/);
  if (fields.length !== 5) return false;
  const t = getTimeParts(date, timeZone);
  return fieldMatches(fields[0], t.minute, 0, 59) && fieldMatches(fields[1], t.hour, 0, 23) &&
    fieldMatches(fields[2], t.day, 1, 31) && fieldMatches(fields[3], t.month, 1, 12) && fieldMatches(fields[4], t.weekday, 0, 6);
}

async function uploadToYouTube(filePath: string, token: string, metadata: { title: string; description: string; tags: string[] }) {
  const size = statSync(filePath).size;
  const init = await fetch("https://www.googleapis.com/upload/youtube/v3/videos?uploadType=resumable&part=snippet,status", {
    method: "POST",
    headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json", "X-Upload-Content-Type": "video/mp4", "X-Upload-Content-Length": String(size) },
    body: JSON.stringify({ snippet: { title: metadata.title.slice(0, 100), description: metadata.description.slice(0, 4900), tags: metadata.tags.slice(0, 20), categoryId: "22" }, status: { privacyStatus: "public", selfDeclaredMadeForKids: false } }),
  });
  const initData = await init.json().catch(() => ({}));
  if (!init.ok) throw new Error(initData?.error?.message || `YouTube upload initialization failed (${init.status})`);
  const uploadUrl = init.headers.get("location");
  if (!uploadUrl) throw new Error("YouTube did not return a resumable upload URL.");

  const source = createReadStream(filePath, { highWaterMark: 1024 * 1024 });
  const iterator = source[Symbol.asyncIterator]();
  const stream = new ReadableStream<Uint8Array>({
    async pull(controller) {
      try { const next = await iterator.next(); if (next.done) controller.close(); else controller.enqueue(next.value as Uint8Array); }
      catch (error) { controller.error(error); }
    },
    async cancel() { source.destroy(); },
  });
  const result = await fetch(uploadUrl, { method: "PUT", headers: { "Content-Type": "video/mp4", "Content-Length": String(size) }, body: stream, duplex: "half" } as RequestInit & { duplex: "half" });
  if (!result.ok) throw new Error(`YouTube upload failed (${result.status}): ${(await result.text()).slice(0, 500)}`);
  const uploaded = await result.json();
  if (!uploaded.id) throw new Error("YouTube did not confirm a video ID.");

}

      const voice = await callJson(baseUrl, "/api/voiceover", { text: scriptData.script, category: settings.category || "general", isShorts: true, translateHindi: settings.language === "hi" });
      const rendered = await callJson(baseUrl, "/api/video/generate", {
        audioUrl: voice.audioUrl, title: scriptData.title, script: scriptData.script, hook: scriptData.hook,
        videoType: "shorts", category: settings.category || "general", language: settings.language,
      });
      const thumb = await callJson(baseUrl, "/api/thumbnail", { title: scriptData.title, boldText: scriptData.thumbnail?.boldText || scriptData.title, category: settings.category || "general", videoType: "shorts" });

      const commonVideoData = {
        title: String(scriptData.title || topic).slice(0, 100), script: String(scriptData.script || ""), hook: String(scriptData.hook || ""),
        description: String(scriptData.description || ""), tags: JSON.stringify(Array.isArray(scriptData.tags) ? scriptData.tags : []),
        videoType: "shorts", audioUrl: String(voice.audioUrl || ""), videoFileUrl: String(rendered.videoUrl || ""),
        thumbnailUrl: String(thumb.thumbnailUrl || ""), subCategory: settings.subCategory,
      };

      if (!settings.isAutoApprove) {
        await prisma.video.update({ where: { id: videoRecord.id }, data: { ...commonVideoData, status: "pending-approval" } });
        results.push({ userId: settings.userId, videoId: videoRecord.id, status: "pending-approval" });
        continue;
      }

      let accessToken: string;
      if (!account.accessTokenExpiry || account.accessTokenExpiry.getTime() < Date.now() + 120000) {
        if (!account.refreshToken) throw new Error("YouTube token expired and no refresh token is stored. Reconnect YouTube.");
        const refreshed = await refreshAccessToken(decryptSocialToken(account.refreshToken));
        accessToken = refreshed.access_token;
        await prisma.socialAccount.update({ where: { id: account.id }, data: { accessToken: encryptSocialToken(accessToken), accessTokenExpiry: new Date(Date.now() + Number(refreshed.expires_in || 3600) * 1000) } });
      } else accessToken = decryptSocialToken(account.accessToken);

      const fileName = path.basename(String(rendered.videoUrl || ""));
      const videoPath = path.resolve(process.cwd(), "storage", "videos", fileName);
      if (!fileName || !videoPath.startsWith(path.resolve(process.cwd(), "storage", "videos") + path.sep) || !existsSync(videoPath)) throw new Error("Rendered video file is missing from local storage.");
      const youtubeVideoId = await uploadToYouTube(videoPath, accessToken, { title: commonVideoData.title, description: commonVideoData.description, tags: JSON.parse(commonVideoData.tags) });
      await prisma.video.update({ where: { id: videoRecord.id }, data: { ...commonVideoData, status: "published", userApproved: true, userApprovedAt: now, youtubeVideoId, youtubeUrl: `https://www.youtube.com/watch?v=${youtubeVideoId}`, uploadedAt: new Date() } });
      results.push({ userId: settings.userId, videoId: videoRecord.id, status: "published", youtubeUrl: `https://www.youtube.com/watch?v=${youtubeVideoId}` });
    } catch (error) {
      const message = error instanceof Error ? error.message : "Unknown autopilot failure";
      if (videoRecord) await prisma.video.update({ where: { id: videoRecord.id }, data: { status: "failed", uploadError: message.slice(0, 1000) } }).catch(() => undefined);
      results.push({ userId: settings.userId, status: "failed", error: message });
    }
  }

  return NextResponse.json({ success: true, checkedAt: now.toISOString(), timeZone, matchedUsers: results.length, results });
}

import { NextResponse } from "next/server";
import { getServerSession } from "next-auth";
import { authOptions } from "@/app/api/auth/[...nextauth]/route";
import { prisma } from "@/lib/prisma";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

const defaults = {
  category: "cinematic",
  subCategory: "sigma",
  cronSchedule: "0 18 * * *",
  isAutoApprove: false,
  language: "en",
  isActive: false,
};

function validCron(value: unknown): value is string {
  if (typeof value !== "string" || value.length > 100) {
    return false;
  }

  const fields = value.trim().split(/\s+/);

  return (
    fields.length === 5 &&
    fields.every((field) => /^[0-9*/,-]+$/.test(field))
  );
}

async function getUserId(): Promise<string | null> {
  const session = await getServerSession(authOptions);

  const userId = (session?.user as { id?: string } | undefined)?.id;

  return typeof userId === "string" && userId.length > 0
    ? userId
    : null;
}

export async function GET() {
  try {
    const userId = await getUserId();

    if (!userId) {
      return NextResponse.json(
        { error: "Please log in first." },
        { status: 401 }
      );
    }

    const settings = await prisma.autopilotSettings.findUnique({
      where: { userId },
    });

    return NextResponse.json({
      settings: settings ?? { ...defaults },
    });
  } catch (error) {
    console.error("Autopilot settings GET error:", error);

    return NextResponse.json(
      { error: "Unable to load autopilot settings." },
      { status: 500 }
    );
  }
}

export async function POST(request: Request) {
  try {
    const userId = await getUserId();

    if (!userId) {
      return NextResponse.json(
        { error: "Please log in first." },
        { status: 401 }
      );
    }

    let body: unknown;

    try {
      body = await request.json();
    } catch {
      return NextResponse.json(
        { error: "Invalid JSON request body." },
        { status: 400 }
      );
    }

    if (
      typeof body !== "object" ||
      body === null ||
      Array.isArray(body)
    ) {
      return NextResponse.json(
        { error: "Request body must be a JSON object." },
        { status: 400 }
      );
    }

    const data = body as Record<string, unknown>;

    const category = typeof data.category === "string" ? data.category.trim().toLowerCase() : defaults.category;

    const subCategory =
      typeof data.subCategory === "string"
        ? data.subCategory.trim()
        : defaults.subCategory;

    const language =
      typeof data.language === "string"
        ? data.language.trim()
        : defaults.language;

    const cronSchedule =
      data.cronSchedule ?? defaults.cronSchedule;

    if (!category || category.length > 50 || !/^[a-z0-9 _-]+$/.test(category)) {
      return NextResponse.json({ error: "Invalid category." }, { status: 400 });
    }

    if (!subCategory || subCategory.length > 80) {
      return NextResponse.json(
        { error: "Invalid subCategory." },
        { status: 400 }
      );
    }

    if (!language || language.length > 20) {
      return NextResponse.json(
        { error: "Invalid language." },
        { status: 400 }
      );
    }

    if (!validCron(cronSchedule)) {
      return NextResponse.json(
        { error: "Schedule must contain five valid-format cron fields." },
        { status: 400 }
      );
    }

    if (
      data.isActive !== undefined &&
      typeof data.isActive !== "boolean"
    ) {
      return NextResponse.json(
        { error: "isActive must be true or false." },
        { status: 400 }
      );
    }

    if (
      data.isAutoApprove !== undefined &&
      typeof data.isAutoApprove !== "boolean"
    ) {
      return NextResponse.json(
        { error: "isAutoApprove must be true or false." },
        { status: 400 }
      );
    }

    const settings = await prisma.autopilotSettings.upsert({
      where: { userId },

      create: {
        userId,
        category,
        subCategory,
        language,
        cronSchedule: cronSchedule.trim(),
        isAutoApprove: data.isAutoApprove ?? false,
        isActive: data.isActive ?? false,
      },

      update: {
        category,
        subCategory,
        language,
        cronSchedule: cronSchedule.trim(),

        ...(data.isAutoApprove !== undefined && {
          isAutoApprove: data.isAutoApprove,
        }),

        ...(data.isActive !== undefined && {
          isActive: data.isActive,
        }),
      },
    });

    return NextResponse.json({
      success: true,
      settings,
    });
  } catch (error) {
    console.error("Autopilot settings POST error:", error);

    return NextResponse.json(
      { error: "Unable to save autopilot settings." },
      { status: 500 }
    );
  }
}
