import { PrismaClient } from "@prisma/client";
import { PrismaLibSQL } from "@prisma/adapter-libsql";
import path from "path";

const globalForPrisma = globalThis as unknown as {
  prisma: PrismaClient | undefined;
};

function createPrismaClient(): PrismaClient {
  const dbPath = path
    .join(process.cwd(), "prisma", "dev.db")
    .replace(/\\/g, "/");

  const adapter = new PrismaLibSQL({
    url: `file:${dbPath}`,
  });

  return new PrismaClient({
    adapter,
    log: ["error"],
  });
}

export const prisma =
  globalForPrisma.prisma ?? createPrismaClient();

if (process.env.NODE_ENV !== "production") {
  globalForPrisma.prisma = prisma;
}

"use client"
import type React from "react"
import { createContext, useContext, useState, useEffect } from "react"
import { signIn, signOut, useSession } from "next-auth/react"

interface User {
  id: string; email: string; name: string; phone: string
  role: string; plan: string; hasCompletedSetup: boolean
  freeVideosUsed: number; paidVideoCredits: number
}

interface AuthContextType {
  user: User | null
  login: (email: string, password: string) => Promise<boolean>
  signup: (name: string, email: string, password: string, phone: string) => Promise<boolean>
  logout: () => void
  isLoading: boolean
}

const AuthContext = createContext<AuthContextType | undefined>(undefined)

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const { data: session, status } = useSession()
  const [extraData, setExtraData] = useState<Partial<User>>({})

  const user: User | null = session?.user ? {
    id: (session.user as any).id || "",
    email: session.user.email || "",
    name: session.user.name || "",
    phone: extraData.phone || "",
    role: (session.user as any).role || "user",
    plan: (session.user as any).plan || "free",
    hasCompletedSetup: extraData.hasCompletedSetup ?? true,
    freeVideosUsed: extraData.freeVideosUsed || 0,
    paidVideoCredits: extraData.paidVideoCredits || 0,
  } : null

  const login = async (email: string, password: string): Promise<boolean> => {
    const result = await signIn("credentials", { email, password, redirect: false })
    return !result?.error
  }

  const signup = async (name: string, email: string, password: string, phone: string): Promise<boolean> => {
    try {
      const res = await fetch("/api/auth/signup", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ name, email, password, phone })
      })
      const data = await res.json()
      if (!data.success) return false
      const result = await signIn("credentials", { email, password, redirect: false })
      return !result?.error
    } catch { return false }
  }

  const logout = () => signOut({ callbackUrl: "/login" })

  return (
    <AuthContext.Provider value={{ user, login, signup, logout, isLoading: status === "loading" }}>
      {children}
    </AuthContext.Provider>
  )
}

export function useAuth() {
  const ctx = useContext(AuthContext)
  if (!ctx) throw new Error("useAuth must be used within AuthProvider")
  return ctx
}

import { NextRequest, NextResponse } from "next/server";
import prisma from "@/lib/db";

export const runtime = "nodejs";

export async function GET(req: NextRequest) {
  try {
    const { searchParams } = new URL(req.url);
    const email = searchParams.get("email");
    const id = searchParams.get("id");

    if (email) {
      return NextResponse.json(
        await prisma.user.findUnique({ where: { email } })
      );
    }

    if (id) {
      return NextResponse.json(
        await prisma.user.findUnique({ where: { id } })
      );
    }

    return NextResponse.json(await prisma.user.findMany());
  } catch (error: any) {
    return NextResponse.json(
      { error: error.message || "Failed to fetch users" },
      { status: 500 }
    );
  }
}

export async function POST(req: NextRequest) {
  try {
    const data = await req.json();

    if (!data.email) {
      return NextResponse.json(
        { error: "Email is required" },
        { status: 400 }
      );
    }

    const user = await prisma.user.upsert({
      where: { email: data.email },
      update: data,
      create: data,
    });

    return NextResponse.json(user);
  } catch (error: any) {
    return NextResponse.json(
      { error: error.message || "Failed to save user" },
      { status: 500 }
    );
  }
}

export async function PATCH(req: NextRequest) {
  try {
    const { id, ...updates } = await req.json();

    if (!id) {
      return NextResponse.json(
        { error: "User ID is required" },
        { status: 400 }
      );
    }

    const user = await prisma.user.update({
      where: { id },
      data: updates,
    });

    return NextResponse.json(user);
  } catch (error: any) {
    return NextResponse.json(
      { error: error.message || "Failed to update user" },
      { status: 500 }
    );
  }
}

import NextAuth, { NextAuthOptions } from "next-auth";
import CredentialsProvider from "next-auth/providers/credentials";
import GoogleProvider from "next-auth/providers/google";
import bcrypt from "bcryptjs";
import { prisma } from "@/lib/prisma";

export const authOptions: NextAuthOptions = {
  secret: process.env.NEXTAUTH_SECRET,
  session: {
    strategy: "jwt",
  },
  pages: {
    signIn: "/login",
  },

  providers: [
    GoogleProvider({
      clientId:
        process.env.GOOGLE_CLIENT_ID ||
        process.env.YOUTUBE_CLIENT_ID ||
        "",
      clientSecret:
        process.env.GOOGLE_CLIENT_SECRET ||
        process.env.YOUTUBE_CLIENT_SECRET ||
        "",
    }),

    CredentialsProvider({
      name: "credentials",

      credentials: {
        email: {
          label: "Email",
          type: "email",
        },
        password: {
          label: "Password",
          type: "password",
        },
      },

      async authorize(credentials) {
        if (
          !credentials?.email ||
          !credentials?.password
        ) {
          return null;
        }

        try {
          const cleanEmail = credentials.email
            .toLowerCase()
            .trim();

          const user = await prisma.user.findUnique({
            where: {
              email: cleanEmail,
            },
          });

          // User must exist and have a password hash.
          if (!user || !user.password) {
            return null;
          }

          const valid = await bcrypt.compare(
            credentials.password,
            user.password
          );

          if (!valid) {
            return null;
          }

          return {
            id: user.id,
            email: user.email,
            name: user.name,
            role: user.role || "user",
            plan: user.plan || "free",
          };
        } catch (error: unknown) {
          console.error(
            "[NEXTAUTH AUTHORIZE ERROR]:",
            error instanceof Error
              ? error.message
              : "Unknown error"
          );

          return null;
        }
      },
    }),
  ],

  callbacks: {
    async signIn({ user, account }) {
      if (account?.provider === "google" && user.email) {
        try {
          const cleanEmail = user.email
            .toLowerCase()
            .trim();

          const existingUser = await prisma.user.findUnique({
            where: {
              email: cleanEmail,
            },
          });

          if (!existingUser) {
            await prisma.user.create({
              data: {
                email: cleanEmail,
                name: user.name || "Google User",
                role: "user",
                plan: "free",
              },
            });
          }
        } catch (error: unknown) {
          console.error(
            "[NEXTAUTH GOOGLE SIGNIN DB ERROR]:",
            error instanceof Error
              ? error.message
              : "Unknown error"
          );
        }
      }

      return true;
    },

    async jwt({ token, user, account }) {
      // Credentials login supplies the database user.
      if (user && account?.provider === "credentials") {
        token.id = user.id;
        token.role = (user as any).role || "user";
        token.plan = (user as any).plan || "free";
      }

      // Resolve Google user's database record.
      if (
        account?.provider === "google" &&
        user?.email
      ) {
        try {
          const dbUser = await prisma.user.findUnique({
            where: {
              email: user.email.toLowerCase().trim(),
            },
          });

          if (dbUser) {
            token.id = dbUser.id;
            token.role = dbUser.role || "user";
            token.plan = dbUser.plan || "free";
          }
        } catch (error) {
          console.error(
            "[NEXTAUTH JWT DB ERROR]:",
            error instanceof Error
              ? error.message
              : "Unknown error"
          );
        }
      }

      // Refresh database-backed role and plan for existing sessions.
      if (!user && token.email) {
        try {
          const dbUser = await prisma.user.findUnique({
            where: {
              email: token.email.toLowerCase().trim(),
            },
          });

          if (dbUser) {
            token.id = dbUser.id;
            token.role = dbUser.role || "user";
            token.plan = dbUser.plan || "free";
          }
        } catch (error) {
          console.error(
            "[NEXTAUTH JWT REFRESH ERROR]:",
            error instanceof Error
              ? error.message
              : "Unknown error"
          );
        }
      }

      return token;
    },

    async session({ session, token }) {
      if (session.user) {
        (session.user as any).id =
          token.id || token.sub;

        (session.user as any).role =
          token.role || "user";

        (session.user as any).plan =
          token.plan || "free";
      }

      return session;
    },
  },
};

const handler = NextAuth(authOptions);

export { handler as GET, handler as POST };

export async function POST(req: NextRequest) {
  try {
    const { email, password } = await req.json();

    if (
      typeof email !== "string" ||
      typeof password !== "string" ||
      !email.trim() ||
      !password
    ) {
      return NextResponse.json(
        { error: "Email and password are required" },
        { status: 400 }
      );
    }

    const cleanEmail = email.toLowerCase().trim();

    const user = await prisma.user.findUnique({
      where: { email: cleanEmail },
    });

    // Reject unknown users and accounts without a password.
    if (!user || !user.password) {
      return NextResponse.json(
        { error: "Invalid email or password" },
        { status: 401 }
      );
    }

    const valid = await bcrypt.compare(password, user.password);

    if (!valid) {
      return NextResponse.json(
        { error: "Invalid email or password" },
        { status: 401 }
      );
    }

    return NextResponse.json({
      success: true,
      user: {
        id: user.id,
        email: user.email,
        name: user.name,
        phone: user.phone || "",
        role: user.role,
        plan: user.plan,
        hasCompletedSetup: !!user.hasCompletedSetup,
        freeVideosUsed: user.freeVideosUsed || 0,
        paidVideoCredits: user.paidVideoCredits || 0,
      },
    });
  } catch (error: unknown) {
    console.error("Login Error:", error);

    return NextResponse.json(
      { error: "Unable to process login right now" },
      { status: 500 }
    );
  }
}

function validCron(value: unknown): value is string {
  if (typeof value !== "string" || value.length > 100) {
    return false;
  }

  const fields = value.trim().split(/\s+/);

  return (
    fields.length === 5 &&
    fields.every((field) => /^[0-9*/,-]+$/.test(field))
  );
}

async function getUserId(): Promise<string | null> {
  const session = await getServerSession(authOptions);

  const userId = (session?.user as { id?: string } | undefined)?.id;

    : null;
}

export async function POST(request: Request) {
  try {
    const userId = await getUserId();

    if (!userId) {
      return NextResponse.json(
        { error: "Please log in first." },
        { status: 401 }
      );
    }

    let body: unknown;

    try {
      body = await request.json();
    } catch {
      return NextResponse.json(
        { error: "Invalid JSON request body." },
        { status: 400 }
      );
    }

    if (
      typeof body !== "object" ||
      body === null ||
      Array.isArray(body)
    ) {
      return NextResponse.json(
        { error: "Request body must be a JSON object." },
        { status: 400 }
      );
    }

    const data = body as Record<string, unknown>;

    const category = typeof data.category === "string" ? data.category.trim().toLowerCase() : defaults.category;

    const subCategory =
      typeof data.subCategory === "string"
        ? data.subCategory.trim()
        : defaults.subCategory;

    const language =
      typeof data.language === "string"
        ? data.language.trim()
        : defaults.language;

    const cronSchedule =
      data.cronSchedule ?? defaults.cronSchedule;

    if (!category || category.length > 50 || !/^[a-z0-9 _-]+$/.test(category)) {
      return NextResponse.json({ error: "Invalid category." }, { status: 400 });
    }

    if (!subCategory || subCategory.length > 80) {
      return NextResponse.json(
        { error: "Invalid subCategory." },
        { status: 400 }
      );
    }

    if (!language || language.length > 20) {
      return NextResponse.json(
        { error: "Invalid language." },
        { status: 400 }
      );
    }

    if (!validCron(cronSchedule)) {
      return NextResponse.json(
        { error: "Schedule must contain five valid-format cron fields." },
        { status: 400 }
      );
    }

    if (
      data.isActive !== undefined &&
      typeof data.isActive !== "boolean"
    ) {
      return NextResponse.json(
        { error: "isActive must be true or false." },
        { status: 400 }
      );
    }

    if (
      data.isAutoApprove !== undefined &&
      typeof data.isAutoApprove !== "boolean"
    ) {
      return NextResponse.json(
        { error: "isAutoApprove must be true or false." },
        { status: 400 }
      );
    }

    const settings = await prisma.autopilotSettings.upsert({
      where: { userId },

      create: {
        userId,
        category,
        subCategory,
        language,
        cronSchedule: cronSchedule.trim(),
        isAutoApprove: data.isAutoApprove ?? false,
        isActive: data.isActive ?? false,
      },

      update: {
        category,
        subCategory,
        language,
        cronSchedule: cronSchedule.trim(),

        ...(data.isAutoApprove !== undefined && {
          isAutoApprove: data.isAutoApprove,
        }),

        ...(data.isActive !== undefined && {
          isActive: data.isActive,
        }),
      },
    });

    return NextResponse.json({
      success: true,
      settings,
    });
  } catch (error) {
    console.error("Autopilot settings POST error:", error);

    return NextResponse.json(
      { error: "Unable to save autopilot settings." },
      { status: 500 }
    );
  }
}

import { NextResponse } from "next/server";
import { createReadStream, existsSync, statSync } from "node:fs";

import { prisma } from "@/lib/prisma";

type CronParts = { minute: number; hour: number; day: number; month: number; weekday: number };

function getTimeParts(date: Date, timeZone: string): CronParts {
  const parts = new Intl.DateTimeFormat("en-US", {
    timeZone, minute: "2-digit", hour: "2-digit", day: "2-digit", month: "2-digit", weekday: "short", hourCycle: "h23",
  }).formatToParts(date);
  const v = Object.fromEntries(parts.map((p) => [p.type, p.value]));
  const days: Record<string, number> = { Sun: 0, Mon: 1, Tue: 2, Wed: 3, Thu: 4, Fri: 5, Sat: 6 };
  return { minute: Number(v.minute), hour: Number(v.hour), day: Number(v.day), month: Number(v.month), weekday: days[v.weekday] ?? date.getDay() };
}

function fieldMatches(field: string, value: number, min: number, max: number): boolean {
  if (field === "*") return true;
  return field.split(",").some((part) => {
    const [base, stepText] = part.split("/");
    const step = stepText ? Number(stepText) : 1;
    if (!Number.isInteger(step) || step < 1) return false;
    let low = min, high = max;
    if (base && base !== "*") {
      if (base.includes("-")) {
        const pair = base.split("-").map(Number);
        if (pair.length !== 2) return false;
        [low, high] = pair;
      } else {
        low = Number(base); high = low;
      }
    }
    return Number.isInteger(low) && Number.isInteger(high) && low >= min && high <= max && value >= low && value <= high && (value - low) % step === 0;
  });
}

function matchesSchedule(expression: string, date: Date, timeZone: string): boolean {
  const fields = expression.trim().split(/\s+/);
  if (fields.length !== 5) return false;
  const t = getTimeParts(date, timeZone);
  return fieldMatches(fields[0], t.minute, 0, 59) && fieldMatches(fields[1], t.hour, 0, 23) &&
    fieldMatches(fields[2], t.day, 1, 31) && fieldMatches(fields[3], t.month, 1, 12) && fieldMatches(fields[4], t.weekday, 0, 6);
}

async function callJson(baseUrl: string, route: string, body: Record<string, unknown>) {
  const response = await fetch(`${baseUrl}${route}`, {
    method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(body), cache: "no-store",
  });
  const data = await response.json().catch(() => ({}));
  if (!response.ok || data.success === false || data.error) throw new Error(data.error || `${route} failed (${response.status})`);
  return data;
}

async function uploadToYouTube(filePath: string, token: string, metadata: { title: string; description: string; tags: string[] }) {
  const size = statSync(filePath).size;
  const init = await fetch("https://www.googleapis.com/upload/youtube/v3/videos?uploadType=resumable&part=snippet,status", {
    method: "POST",
    headers: { Authorization: `Bearer ${token}`, "Content-Type": "application/json", "X-Upload-Content-Type": "video/mp4", "X-Upload-Content-Length": String(size) },
    body: JSON.stringify({ snippet: { title: metadata.title.slice(0, 100), description: metadata.description.slice(0, 4900), tags: metadata.tags.slice(0, 20), categoryId: "22" }, status: { privacyStatus: "public", selfDeclaredMadeForKids: false } }),
  });
  const initData = await init.json().catch(() => ({}));
  if (!init.ok) throw new Error(initData?.error?.message || `YouTube upload initialization failed (${init.status})`);
  const uploadUrl = init.headers.get("location");
  if (!uploadUrl) throw new Error("YouTube did not return a resumable upload URL.");

  const source = createReadStream(filePath, { highWaterMark: 1024 * 1024 });
  const iterator = source[Symbol.asyncIterator]();
  const stream = new ReadableStream<Uint8Array>({
    async pull(controller) {
      try { const next = await iterator.next(); if (next.done) controller.close(); else controller.enqueue(next.value as Uint8Array); }
      catch (error) { controller.error(error); }
    },
    async cancel() { source.destroy(); },
  });
  const result = await fetch(uploadUrl, { method: "PUT", headers: { "Content-Type": "video/mp4", "Content-Length": String(size) }, body: stream, duplex: "half" } as RequestInit & { duplex: "half" });
  if (!result.ok) throw new Error(`YouTube upload failed (${result.status}): ${(await result.text()).slice(0, 500)}`);
  const uploaded = await result.json();
  if (!uploaded.id) throw new Error("YouTube did not confirm a video ID.");
  return String(uploaded.id);
}

export async function POST(request: Request) {
  const secret = process.env.CRON_SECRET;
  if (!secret) return NextResponse.json({ error: "CRON_SECRET is not configured." }, { status: 503 });
  if (request.headers.get("authorization") !== `Bearer ${secret}`) return NextResponse.json({ error: "Unauthorized." }, { status: 401 });

  const now = new Date();
  const timeZone = process.env.AUTOPILOT_TIMEZONE || "Asia/Kolkata";
  const baseUrl = new URL(request.url).origin;
  const settingsList = await prisma.autopilotSettings.findMany({ where: { isActive: true } });
  const results: Array<Record<string, unknown>> = [];

  for (const settings of settingsList) {
    if (!matchesSchedule(settings.cronSchedule, now, timeZone)) continue;
    let videoRecord: { id: string } | null = null;
    try {
      const minuteStart = new Date(now); minuteStart.setSeconds(0, 0);
      const duplicate = await prisma.video.findFirst({ where: { userId: settings.userId, createdAt: { gte: minuteStart } }, select: { id: true } });
      if (duplicate) { results.push({ userId: settings.userId, status: "skipped_duplicate_minute" }); continue; }

      const channel = await prisma.channel.findFirst({ where: { userId: settings.userId, isActive: true }, orderBy: { createdAt: "asc" } });
      if (!channel) { results.push({ userId: settings.userId, status: "skipped_no_channel" }); continue; }
      const account = await prisma.socialAccount.findFirst({ where: { userId: settings.userId, provider: "youtube", isConnected: true }, orderBy: { updatedAt: "desc" } });
      if (!account) { results.push({ userId: settings.userId, status: "skipped_youtube_not_connected" }); continue; }

      const topic = `${settings.subCategory} â€” ${new Intl.DateTimeFormat("en-CA", { timeZone }).format(now)} episode`;
      videoRecord = await prisma.video.create({ data: { userId: settings.userId, channelId: channel.id, title: `Generating: ${topic}`.slice(0, 100), topic, status: "autopilot-processing", scheduledDate: now } });

      const scriptData = await callJson(baseUrl, "/api/videos/generate", {
        topic, language: settings.language === "hi" ? "Hindi" : settings.language === "hinglish" ? "Hinglish" : "English",
        channelName: channel.name, category: settings.category || channel.category || "general", mode: "shorts",
      });
      const voice = await callJson(baseUrl, "/api/voiceover", { text: scriptData.script, category: settings.category || "general", isShorts: true, translateHindi: settings.language === "hi" });
      const rendered = await callJson(baseUrl, "/api/video/generate", {
        audioUrl: voice.audioUrl, title: scriptData.title, script: scriptData.script, hook: scriptData.hook,
        videoType: "shorts", category: settings.category || "general", language: settings.language,
      });
      const thumb = await callJson(baseUrl, "/api/thumbnail", { title: scriptData.title, boldText: scriptData.thumbnail?.boldText || scriptData.title, category: settings.category || "general", videoType: "shorts" });

      const commonVideoData = {
        title: String(scriptData.title || topic).slice(0, 100), script: String(scriptData.script || ""), hook: String(scriptData.hook || ""),
        description: String(scriptData.description || ""), tags: JSON.stringify(Array.isArray(scriptData.tags) ? scriptData.tags : []),
        videoType: "shorts", audioUrl: String(voice.audioUrl || ""), videoFileUrl: String(rendered.videoUrl || ""),
        thumbnailUrl: String(thumb.thumbnailUrl || ""), subCategory: settings.subCategory,
      };

      if (!settings.isAutoApprove) {
        await prisma.video.update({ where: { id: videoRecord.id }, data: { ...commonVideoData, status: "pending-approval" } });
        results.push({ userId: settings.userId, videoId: videoRecord.id, status: "pending-approval" });
        continue;
      }

      let accessToken: string;
      if (!account.accessTokenExpiry || account.accessTokenExpiry.getTime() < Date.now() + 120000) {
        if (!account.refreshToken) throw new Error("YouTube token expired and no refresh token is stored. Reconnect YouTube.");
        const refreshed = await refreshAccessToken(decryptSocialToken(account.refreshToken));
        accessToken = refreshed.access_token;
        await prisma.socialAccount.update({ where: { id: account.id }, data: { accessToken: encryptSocialToken(accessToken), accessTokenExpiry: new Date(Date.now() + Number(refreshed.expires_in || 3600) * 1000) } });
      } else accessToken = decryptSocialToken(account.accessToken);

      const fileName = path.basename(String(rendered.videoUrl || ""));
      const videoPath = path.resolve(process.cwd(), "storage", "videos", fileName);
      if (!fileName || !videoPath.startsWith(path.resolve(process.cwd(), "storage", "videos") + path.sep) || !existsSync(videoPath)) throw new Error("Rendered video file is missing from local storage.");
      const youtubeVideoId = await uploadToYouTube(videoPath, accessToken, { title: commonVideoData.title, description: commonVideoData.description, tags: JSON.parse(commonVideoData.tags) });
      await prisma.video.update({ where: { id: videoRecord.id }, data: { ...commonVideoData, status: "published", userApproved: true, userApprovedAt: now, youtubeVideoId, youtubeUrl: `https://www.youtube.com/watch?v=${youtubeVideoId}`, uploadedAt: new Date() } });
      results.push({ userId: settings.userId, videoId: videoRecord.id, status: "published", youtubeUrl: `https://www.youtube.com/watch?v=${youtubeVideoId}` });
    } catch (error) {
      const message = error instanceof Error ? error.message : "Unknown autopilot failure";
      if (videoRecord) await prisma.video.update({ where: { id: videoRecord.id }, data: { status: "failed", uploadError: message.slice(0, 1000) } }).catch(() => undefined);
      results.push({ userId: settings.userId, status: "failed", error: message });
    }
  }

  return NextResponse.json({ success: true, checkedAt: now.toISOString(), timeZone, matchedUsers: results.length, results });
}

type FilterType = "pending" | "approved" | "rejected"

interface VideoApprovalListProps {
  filter: FilterType
}

interface UploadResult {
  success?: boolean
  queued?: boolean
  error?: string
  message?: string
  videoId?: string
  youtubeUrl?: string
  status?: string
}

export function VideoApprovalList({ filter }: VideoApprovalListProps) {
  const { user } = useAuth()
  const [videos, setVideos] = useState<VideoType[]>([])
  const [loading, setLoading] = useState(true)
  const [busyId, setBusyId] = useState<string | null>(null)
  const [message, setMessage] = useState("")
  const [error, setError] = useState("")

  const refreshVideos = useCallback(() => {
    try {
      setVideos(store ? store.getVideos() : [])
    } catch (e) {
      console.error("Could not load videos:", e)
      setError("Videos load nahi ho paaye.")
    } finally {
      setLoading(false)
    }
  }, [])

  useEffect(() => {
    refreshVideos()

    const onStorage = () => refreshVideos()
    window.addEventListener("storage", onStorage)

    return () => window.removeEventListener("storage", onStorage)
  }, [refreshVideos])

  const filteredVideos = useMemo(() => {
    return videos
      .filter((video) => {
        if (filter === "pending") {
          return (
            video.status === "pending-approval" ||
            (!video.adminApproved &&
              video.status !== "rejected" &&
              video.status !== "live" &&
              video.status !== "failed")
          )
        }

        if (filter === "approved") {
          return video.adminApproved === true && video.status !== "rejected"
        }

        return video.status === "rejected"
      })
      .sort(
        (a, b) =>
          new Date(b.createdAt || 0).getTime() -
          new Date(a.createdAt || 0).getTime(),
      )
  }, [videos, filter])

  const handleApprove = async (video: VideoType) => {
    if (!store || !user?.id) {
      setError("Admin session ya local video store available nahi hai.")
      return
    }

    if (!video.videoUrl) {
      setError(
        `"${video.title}" ki video file missing hai. Pehle video generate karke save karo.`,
      )
      return
    }

    if (!video.thumbnailUrl && !video.thumbnail) {
      setError(
        `"${video.title}" ka thumbnail missing hai. Pehle thumbnail generate karo.`,
      )
      return
    }

    if (video.status === "live" && video.youtubeUrl) {
      setMessage("Yeh video pehle se YouTube par publish ho chuka hai.")
      return
    }

    const confirmed = window.confirm(
      `"${video.title}" ko approve karke YouTube par PUBLIC publish karna hai? Agar upload successful hua, video public ho sakta hai.`,
    )

    if (!confirmed) return

    setBusyId(video.id)
    setError("")
    setMessage("YouTube par video upload ho raha hai. Is window ko band mat karo.")

    try {
      // Save admin approval first. Upload success is tracked separately.
      store.adminApproveVideo(video.id, user.id)
      refreshVideos()

      const response = await fetch("/api/youtube/upload", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          videoUrl: video.videoUrl,
          title: video.title,
          description: video.description || "",
          tags: video.tags || [],
          categoryId: "22",
          privacyStatus: "public",
          language: "hi",
        }),
      })

      const result = (await response.json().catch(() => ({}))) as UploadResult

      if (!response.ok || !result.success) {
        if (result.queued) {
          setMessage(
            result.message ||
              "Video upload queue mein add hua hai; abhi Live nahi hai.",
          )
          refreshVideos()
          return
        }

        throw new Error(result.error || result.message || `Upload failed (${response.status})`)
      }

      if (!result.videoId || !result.youtubeUrl) {
        throw new Error(
          "Upload API ne success response diya, lekin YouTube video ID/link nahi mila. YouTube par status verify karo.",
        )
      }

      // Mark Live only when the API confirms public status.
      if (result.status?.toLowerCase() === "public") {
        store.updateVideo(video.id, {
          status: "live",
          youtubeUrl: result.youtubeUrl,
          adminApproved: true,

          `Video YouTube par publish ho gaya: ${result.youtubeUrl}`,

          youtubeUrl: result.youtubeUrl,
          adminApproved: true,
          status: "approved",

        )

    } finally {

                          <img
                            src={thumbnail}
                            alt={`Thumbnail: ${video.title}`}
                            className="aspect-video w-full object-cover"
                            onError={(event) => {
                              event.currentTarget.style.display = "none"
                            }}
                          />
                        </a>
                      ) : (
                        <div className="flex aspect-video items-center justify-center text-sm text-white">
                          Thumbnail unavailable
                        </div>
                      )}
                    </div>

                    {videoUrl ? (
                      <video
                        key={videoUrl}
                        src={videoUrl}
                        controls
                        preload="metadata"
                        className="aspect-video w-full rounded-lg border bg-black"
                      >
                        Your browser does not support video playback.
                      </video>
                    ) : (
                      <div className="flex aspect-video items-center justify-center rounded-lg border border-dashed text-center text-sm text-muted-foreground">
                        <div>
                          <VideoIcon className="mx-auto mb-2 h-7 w-7" />
                          Actual video file missing
                        </div>
                      </div>
                    )}
                  </div>

                  <div className="min-w-0 space-y-4">
                    <div className="flex flex-wrap items-start justify-between gap-3">
                      <div className="min-w-0">
                        <h2 className="break-words text-xl font-bold">
                          {video.title || "Untitled video"}
                        </h2>

                        <p className="mt-1 break-all text-xs text-muted-foreground">
                          ID: {video.id}
                        </p>
                      </div>

                      <Badge
                        variant="outline"
                        className={
                          video.status === "rejected"
                            ? "border-red-500/40 text-red-600"
                            : video.status === "live"
                              ? "border-emerald-500/40 text-emerald-600"
                              : video.adminApproved
                                ? "border-blue-500/40 text-blue-600"
                                : "border-amber-500/40 text-amber-600"
                        }
                      >
                        {status === "Live" ? (
                          <CheckCircle2 className="mr-1 h-3 w-3" />
                        ) : status === "Rejected" ? (
                          <XCircle className="mr-1 h-3 w-3" />
                        ) : (
                          <Clock3 className="mr-1 h-3 w-3" />
                        )}
                        {status}
                      </Badge>
                    </div>

                    <div>
                      <h3 className="mb-1 text-sm font-semibold">
                        Description
                      </h3>
                      <p className="whitespace-pre-wrap break-words text-sm text-muted-foreground">
                        {video.description || "No description provided."}
                      </p>
                    </div>

                    <div>
                      <h3 className="mb-1 text-sm font-semibold">Tags</h3>
                      <div className="flex flex-wrap gap-2">
                        {video.tags?.length ? (
                          video.tags.map((tag, index) => (
                            <Badge
                              key={`${tag}-${index}`}
                              variant="secondary"
                            >
                              {tag}
                            </Badge>
                          ))
                        ) : (
                          <span className="text-sm text-muted-foreground">
                            No tags
                          </span>
                        )}
                      </div>
                    </div>

                    {video.script && (
                      <details className="rounded-lg border p-3">
                        <summary className="cursor-pointer text-sm font-semibold">
                          View script
                        </summary>
                        <p className="mt-3 whitespace-pre-wrap text-sm text-muted-foreground">
                          {video.script}
                        </p>
                      </details>
                    )}

                    <div className="text-xs text-muted-foreground">
                      Created:{" "}
                      {video.createdAt
                        ? new Date(video.createdAt).toLocaleString()
                        : "Unknown"}
                    </div>

                    {video.youtubeUrl && (
                      <a
                        href={video.youtubeUrl}
                        target="_blank"
                        rel="noreferrer"
                        className="inline-flex items-center gap-2 break-all text-sm text-blue-600 underline"
                      >
                        View on YouTube
                        <ExternalLink className="h-3 w-3 shrink-0" />
                      </a>
                    )}

                    {filter === "pending" && (
                      <div className="flex flex-wrap gap-3 border-t pt-4">
                        <Button
                          onClick={() => handleApprove(video)}
                          disabled={busyId === video.id}
                          className="bg-emerald-600 text-white hover:bg-emerald-700"
                        >
                          {busyId === video.id ? (
                            <RefreshCw className="mr-2 h-4 w-4 animate-spin" />
                          ) : (
                            <CheckCircle2 className="mr-2 h-4 w-4" />
                          )}
                          {busyId === video.id
                            ? "Uploading..."
                            : "Approve & Publish"}
                        </Button>

                        <Button
                          variant="destructive"
                          onClick={() => handleReject(video)}
                          disabled={busyId === video.id}
                        >
                          <XCircle className="mr-2 h-4 w-4" />
                          Reject Video
                        </Button>
                      </div>
                    )}
                  </div>
                </div>
              </article>
            )
          })}
        </div>
      )}
    </div>
  )
}
